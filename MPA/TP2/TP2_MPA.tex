\subsubsection{1.}\label{section}

\begin{itemize}
\tightlist
\item
  Soit \(c=1\). L'expression de la loi générative
  \(p(y|c=1, \theta_1, \theta_2)\) est donnée par:
\end{itemize}

\[
p(y|c=1, \theta_1, \theta_2)=p(y_1y_2...y_n|c=1, \theta_1, \theta_2)
\]

\begin{itemize}
\tightlist
\item
  Comme les \(y_i\) sont indépendants conditionnellement à
  \(\theta=(c, \theta_1, \theta_2)\), on a :
\end{itemize}

\[
\begin{align}
p(y|c=1, \theta_1, \theta_2) &= \prod_{i=1}^n p(y_i|c=1, \theta_1, \theta2) \\
     &= \prod_{i=1}^n \theta_2^{y_i}.(1-\theta_2)^{1-y_i} \\
     &= \theta_2^{\sum_{i=1}^n y_i}.(1-\theta_2)^{n-\sum_{i=1}^n y_i}
\end{align}
\]

\subsubsection{2.}\label{section-1}

\begin{itemize}
\tightlist
\item
  Soit \(c>1\). L'expression de la loi générative
  \(p(y|c, \theta_1, \theta_2)\) est donnée par:
\end{itemize}

\[
\begin{align}
p(y|c=1, \theta_1, \theta_2)&=p(y_1y_2...y_n|c=1, \theta_1, \theta_2) \\
     &= \prod_{i=1}^n p(y_i|c, \theta_1, \theta2) \\
     &= \prod_{i=1}^{c-1} \theta_1^{y_i}.(1-\theta_1)^{1-y_i} . \prod_{i=c}^n \theta_2^{y_i}.(1-\theta_2)^{1-y_i}\\
     &= \theta_1^{\sum_{i=1}^{c-1} y_i}.(1-\theta_1)^{c-1-\sum_{i=1}^{c-1} y_i} . \theta_2^{\sum_{i=c}^n y_i}.(1-\theta_2)^{n-c+1-\sum_{i=c}^n y_i}
\end{align}
\]

\subsubsection{3.}\label{section-2}

\begin{itemize}
\tightlist
\item
  On suppose \(\theta_1\) et \(\theta_2\) connues, on a
  \(\forall  c = 2, ..., n\) : \[
  \begin{align}
  \frac{p(c|y)}{p(c=1|y)} &= \frac{\prod_{i=1}^{c-1} \theta_1^{y_i}.(1-\theta_1)^{1-y_i} . \prod_{i=c}^n \theta_2^{y_i}.(1-\theta_2)^{1-y_i}}{\prod_{i=1}^n \theta_2^{y_i}.(1-\theta_2)^{1-y_i}} \\
                        &= \frac{\prod_{i=1}^{c-1} \theta_1^{y_i}.(1-\theta_1)^{1-y_i}}{\prod_{i=1}^{c-1} \theta_2^{y_i}.(1-\theta_2)^{1-y_i}} \\
                        &= \prod_{i=1}^{c-1} \frac{\theta_1^{y_i}.(1-\theta_1)^{1-y_i}}{\theta_2^{y_i}.(1-\theta_2)^{1-y_i}}
  \end{align}
  \]
\item
  Pour \(c = 2, ...,n\) le rapport \(\frac{p(c|y)}{p(c=1|y)}\) nécessite
  l'ordre de \(c-1\) multiplications (pour \(c = 1\) aucune
  multiplication n'est nécessaire) d'où le calcule du rapport
  \(\frac{p(c|y)}{p(c=1|y)}\) pour tout \(c\) se fait en effectuant
  l'ordre de \(\sum_{c=1}^n c = \frac{n.(n-1)}2\) multiplications.
\end{itemize}

\subsubsection{4.}\label{section-3}

\begin{itemize}
\item
  On note \(R_c = \frac{p(c|y)}{p(c=1|y)}\) et \(S = \sum_{c=1}^n R_c\)
\item
  On a \(\forall c>1\) :
  \(R_c = R_{c-1}.\frac{\theta_1^{y_{c-1}}.(1-\theta_1)^{1-y_{c-1}}}{\theta_2^{y_{c-1}}.(1-\theta_2)^{1-y_{c-1}}}\)
  soit \(3\) multiplications de plus pour chaque itération.
\end{itemize}

\end{itemize}

\subsubsection{5.}\label{section-4}

\begin{itemize}
\item
  Pour tester la validiter de l'algorithme de Gibbs on peut calculer la
  valeur theorique de la probabilité de \(c|y\) :
\item
  On a : \[
  \begin{align}
  p(c|y) &= \int_0^1\int_0^1p(c,\theta_1, \theta_2|y)d\theta_1d\theta_2
  \end{align}
  \] avec : \[
  \begin{align}
  p(c,\theta_1, \theta_2|y) &= p(y|c,\theta_1, \theta_2).\frac{p(\theta)}{p(y)}\\
                  &\propto p(y|c,\theta_1, \theta_2).\frac1n \\
                  &\propto \theta_1^{\sum_{i=1}^{c-1} y_i}.(1-\theta_1)^{c-1-\sum_{i=1}^{c-1} y_i} . \theta_2^{\sum_{i=c}^n y_i}.(1-\theta_2)^{n-c+1-\sum_{i=c}^n y_i} \\
  \end{align}
  \]
\item
  On note \(s_1 = \sum_{i=1}^{c-1} y_i\), \(n_1 = c-1\),
  \(s_2 = \sum_{i=c}^n y_i\) et \(n_2 = n - c +1\) et \(A(y)\) le
  coefficient de proportionalité. Alors on a : \[
  \begin{align}
  p(c|y) &= \int_0^1\int_0^1A(y).\theta_1^{s_1}.(1-\theta_1)^{n_1 - s_1} . \theta_2^{s_2}.(1-\theta_2)^{n_2 - s_2}d\theta_1d\theta_2 \\
       &= A(y).\int_0^1 \theta_1^{s_1}.(1-\theta_1)^{n_1 - s_1}d\theta_1\int_0^1\theta_2^{s_2}.(1-\theta_2)^{n_2 - s_2}d\theta_2 \\
       &= A(y).B(s_1+1, n_1 - s_1 + 1).B(s_2 + 1, n_2 - s_2 + 1)
  \end{align}
  \]
\item
  On se base pour l'algorithm de Gibbs sur la formule:
\end{itemize}

\[
\begin{align}
p(c|y) &= \int_0^1\int_0^1p(c|\theta_1, \theta_2, y)p(\theta_1, \theta_2|y)d\theta_1d\theta_2
\end{align}
\] - On remarque bien que les \(\theta_1|y\) et \(\theta_2|y\) ne sont
pas indépendants.

\begin{itemize}
\tightlist
\item
  On approche donc \(p(c|y)\) par la loi de
  \(p(c|\theta_1, \theta_2, y)\) calculée auparavant pour des valeurs
  ``mises à jour'' de \(\theta_1\) et \(\theta_2\).
\end{itemize}
